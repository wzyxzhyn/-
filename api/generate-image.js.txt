// api/generate-image.js
// 豆包AI图像生成后端代理 - 密钥存在环境变量里，不会暴露给前端

export default async function handler(req, res) {
  // 允许跨域
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
  
  if (req.method === 'OPTIONS') {
    return res.status(200).end();
  }

  if (req.method !== 'POST') {
    return res.status(405).json({ error: '只允许POST请求' });
  }

  const { imageBase64, prompt } = req.body;

  // 从环境变量读取密钥（在Vercel后台配置）
  const VOLC_AK = process.env.VOLC_AK;
  const VOLC_SK = process.env.VOLC_SK;

  if (!VOLC_AK || !VOLC_SK) {
    return res.status(500).json({ error: '未配置火山引擎API密钥，请在Vercel环境变量中设置VOLC_AK和VOLC_SK' });
  }

  try {
    // 调用火山引擎豆包图像生成API
    // 注意：具体API地址和参数以火山引擎官方文档最新版本为准
    // 文档地址：https://www.volcengine.com/docs/6791
    const response = await fetch('https://visual.volcengineapi.com/', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${VOLC_AK}:${VOLC_SK}`
      },
      body: JSON.stringify({
        model: 'seedream-3.0-t2i',
        prompt: prompt,
        image: imageBase64,
        width: 1024,
        height: 1024,
        return_url: true
      })
    });

    const data = await response.json();
    
    if (data.code !== 0 && !data.data) {
      return res.status(500).json({ error: 'AI生成失败', detail: data });
    }

    return res.status(200).json({ 
      imageUrl: data.data?.image_url || data.data?.url 
    });

  } catch (error) {
    return res.status(500).json({ error: '服务器错误', detail: error.message });
  }
}
