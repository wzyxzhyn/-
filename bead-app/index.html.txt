      { name: "黑色", hex: "#000000" }, { name: "深灰", hex: "#333333" },

    // ========== 豆包AI预处理调用 ==========
    async function callDoubaoAI(imgBase64) {
      const prompt = "把这张照片转为像素风Q版卡通，黑色粗轮廓，高饱和度纯色块，纯白色背景，拼豆板104钉方格大小，每个元素用完整格子呈现，边缘必须用完整格子";
      try {
        const res = await fetch('/api/generate-image', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ imageBase64, prompt })
        });
        const data = await res.json();
        if (data.imageUrl) {
          const img = new Image();
          img.crossOrigin = 'anonymous';
          img.src = data.imageUrl;
          await new Promise(r => img.onload = r);
          return img;
        }
      } catch (e) {
        console.warn('AI调用失败，使用本地算法:', e);
      }
      return null;
    }

    // 本地边缘强化
    function localEdgeEnhance(image, strength) {
      const c = document.createElement('canvas');
      const t = c.getContext('2d');
      c.width = image.width; c.height = image.height;
      t.drawImage(image, 0, 0);
      const d = t.getImageData(0, 0, c.width, c.height).data;
      const out = new Uint8ClampedArray(d);
      const w = c.width, h = c.height;
      for (let y = 1; y < h - 1; y++) {
        for (let x = 1; x < w - 1; x++) {
          const i = (y * w + x) * 4;
          const up = ((y-1)*w+x)*4, down = ((y+1)*w+x)*4;
          const left = (y*w+x-1)*4, right = (y*w+x+1)*4;
          const diff = Math.abs(d[i]-d[up])+Math.abs(d[i]-d[down])+Math.abs(d[i]-d[left])+Math.abs(d[i]-d[right])
                     + Math.abs(d[i+1]-d[up+1])+Math.abs(d[i+1]-d[down+1])+Math.abs(d[i+1]-d[left+1])+Math.abs(d[i+1]-d[right+1])
                     + Math.abs(d[i+2]-d[up+2])+Math.abs(d[i+2]-d[down+2])+Math.abs(d[i+2]-d[left+2])+Math.abs(d[i+2]-d[right+2]);
          if (diff > strength * 3) { out[i]=0; out[i+1]=0; out[i+2]=0; }
        }
      }
      t.putImageData(new ImageData(out, w, h), 0, 0);
      const img = new Image();
      img.src = c.toDataURL();
      return img;
    }

    function init() {
      updateCanvasSize(32);
      buildColorPalette();
      bindEvents();
      drawPlaceholder();
    }

    function drawPlaceholder() {
      ctx.fillStyle = '#FFFFFF';
      ctx.fillRect(0, 0, previewCanvas.width, previewCanvas.height);
      ctx.strokeStyle = '#E5E7EB';
      const cell = previewCanvas.width / currentGridSize;
      for (let i = 0; i <= currentGridSize; i++) {
        ctx.beginPath(); ctx.moveTo(i*cell, 0); ctx.lineTo(i*cell, previewCanvas.height); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(0, i*cell); ctx.lineTo(previewCanvas.width, i*cell); ctx.stroke();
      }
    }

    function updateCanvasSize(size) {
      currentGridSize = size;
      canvasSizeText.textContent = size;
      const maxBoard = parseInt(boardSizeSelect.value);
      const cell = Math.max(4, Math.floor(600 / Math.min(size, maxBoard)));
      previewCanvas.width = size * cell;
      previewCanvas.height = size * cell;
      drawPlaceholder();
      if (beadGrid.length > 0) drawBeadGrid();
    }

    function buildColorPalette() {
      colorPalette.innerHTML = '';
      beadColorLibrary.slice(0, currentColorCount).forEach((color, i) => {
        const btn = document.createElement('button');
        btn.className = 'w-10 h-10 rounded border border-gray-300 cursor-pointer';
        btn.style.backgroundColor = color.hex;
        btn.title = `${i+1} ${color.name}`;
        btn.addEventListener('click', () => { selectedColorInput.value = color.hex; });
        colorPalette.appendChild(btn);
      });
    }

    function bindEvents() {
      uploadInput.addEventListener('change', handleUpload);
      canvasSizeSlider.addEventListener('input', e => updateCanvasSize(parseInt(e.target.value)));
      boardSizeSelect.addEventListener('change', () => updateCanvasSize(currentGridSize));
      cutoutToleranceSlider.addEventListener('input', e => { cutoutToleranceText.textContent = e.target.value; if (originalImage) autoCutout(); });
      edgeStrengthSlider.addEventListener('input', e => { edgeStrengthText.textContent = e.target.value; if (originalImage) autoCutout(); });
      autoCutoutBtn.addEventListener('click', autoCutout);
      generateBtn.addEventListener('click', generateBeadPattern);
      undoBtn.addEventListener('click', undo);
      mirrorBtn.addEventListener('click', mirrorPattern);
      export4kBtn.addEventListener('click', export4K);
      addTextBtn.addEventListener('click', addDraggableText);

      document.querySelectorAll('.color-count-btn').forEach(btn => {
        btn.addEventListener('click', e => { currentColorCount = parseInt(e.target.dataset.colors); buildColorPalette(); });
      });
      document.querySelectorAll('.tool-btn').forEach(btn => {
        btn.addEventListener('click', e => {
          currentTool = e.target.dataset.tool;
          document.querySelectorAll('.tool-btn').forEach(b => b.classList.remove('bg-gray-100'));
          e.target.classList.add('bg-gray-100');
        });
      });
      previewCanvas.addEventListener('mousedown', onMouseDown);
      previewCanvas.addEventListener('mousemove', onMouseMove);
      previewCanvas.addEventListener('mouseup', onMouseUp);
    }

    function handleUpload(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = event => {
        const img = new Image();
        img.onload = () => { originalImage = img; autoCutout(); };
        img.src = event.target.result;
      };
      reader.readAsDataURL(file);
    }

    async function autoCutout() {
      if (!originalImage) return;
      const tolerance = parseInt(cutoutToleranceSlider.value);
      const edgeStrength = parseInt(edgeStrengthSlider.value);

      // 先决定用AI还是本地算法
      let srcImg = originalImage;
      if (useAiPreprocess.checked) {
        const base64 = originalImage.src;
        const aiImg = await callDoubaoAI(base64);
        if (aiImg) srcImg = aiImg;
      }
      // 本地轮廓强化（无论是否用AI都叠加一层）
      srcImg = localEdgeEnhance(srcImg, edgeStrength);

      const c = document.createElement('canvas');
      const t = c.getContext('2d');
      c.width = srcImg.width; c.height = srcImg.height;
      t.drawImage(srcImg, 0, 0);
      const imgData = t.getImageData(0, 0, c.width, c.height);
      const d = imgData.data;
      for (let i = 0; i < d.length; i += 4) {
        if (Math.abs(d[i]-255)<=tolerance && Math.abs(d[i+1]-255)<=tolerance && Math.abs(d[i+2]-255)<=tolerance) {
          d[i+3] = 0;
        }
      }
      t.putImageData(imgData, 0, 0);
      processedImage = new Image();
      processedImage.src = c.toDataURL();
      processedImage.onload = () => generateBeadPattern();
    }

    function generateBeadPattern() {
      saveHistory();
      if (!processedImage && !originalImage) return;
      const src = processedImage || originalImage;
      const c = document.createElement('canvas');
      const t = c.getContext('2d');
      const size = currentGridSize;
      c.width = size; c.height = size;
      t.fillStyle = '#FFFFFF';
      t.fillRect(0, 0, size, size);
      const ratio = src.width / src.height;
      let dw, dh, ox, oy;
      if (ratio > 1) { dw = size; dh = size / ratio; ox = 0; oy = (size-dh)/2; }
      else { dh = size; dw = size * ratio; ox = (size-dw)/2; oy = 0; }
      t.drawImage(src, ox, oy, dw, dh);
      const d = t.getImageData(0, 0, size, size).data;
      beadGrid = []; colorMap = {}; colorIndex = 1;
      for (let y = 0; y < size; y++) {
        beadGrid[y] = [];
        for (let x = 0; x < size; x++) {
          const i = (y*size+x)*4;
          if (d[i+3] < 128) { beadGrid[y][x] = null; continue; }
          const hex = '#' + ((1<<24)+(d[i]<<16)+(d[i+1]<<8)+d[i+2]).toString(16).slice(1).toUpperCase();
          beadGrid[y][x] = getColorIndex(hex);
        }
      }
      drawBeadGrid();
      updateLegend();
    }

    function getColorIndex(hex) {
      if (!colorMap[hex]) colorMap[hex] = colorIndex++;
      return colorMap[hex];
    }

    function drawBeadGrid() {
      const size = currentGridSize;
      const cell = previewCanvas.width / size;
      ctx.fillStyle = '#FFFFFF';
      ctx.fillRect(0, 0, previewCanvas.width, previewCanvas.height);
      for (let y = 0; y < size; y++) {
        for (let x = 0; x < size; x++) {
          const id = beadGrid[y][x];
          if (id === null) continue;
          const hex = Object.keys(colorMap).find(k => colorMap[k] === id);
          ctx.fillStyle = hex || '#CCC';
          ctx.fillRect(x*cell, y*cell, cell, cell);
          ctx.strokeStyle = '#000';
          ctx.lineWidth = Math.max(1, cell/16);
          ctx.strokeRect(x*cell, y*cell, cell, cell);
          if (cell > 12) {
            ctx.fillStyle = '#000';
            ctx.font = `${Math.max(8, cell/3)}px Arial`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(id, x*cell+cell/2, y*cell+cell/2);
          }
        }
      }
      // 网格数字标注
      ctx.fillStyle = '#666';
      ctx.font = `${Math.max(10, cell/4)}px Arial`;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'top';
      for (let i = 0; i < size; i++) {
        ctx.fillText(i+1, i*cell+cell/2, 2);
        ctx.fillText(i+1, 2, i*cell+cell/2);
      }
    }

    function updateLegend() {
      legendPanel.innerHTML = '';
      Object.entries(colorMap).forEach(([hex, id]) => {
        const color = beadColorLibrary.find(c => c.hex.toLowerCase() === hex.toLowerCase());
        const item = document.createElement('div');
        item.className = 'flex items-center gap-2';
        item.innerHTML = `<div class="w-5 h-5 rounded border" style="background:${hex}"></div><span>${id} ${color?.name || hex}</span>`;
        legendPanel.appendChild(item);
      });
    }

    function saveHistory() {
      historyStack.push(JSON.stringify(beadGrid));
      if (historyStack.length > 50) historyStack.shift();
    }

    function undo() {
      if (historyStack.length === 0) return;
      beadGrid = JSON.parse(historyStack.pop());
      drawBeadGrid();
    }

    function mirrorPattern() {
      saveHistory();
      const size = currentGridSize;
      for (let y = 0; y < size; y++) {
        for (let x = 0; x < size/2; x++) {
          [beadGrid[y][x], beadGrid[y][size-1-x]] = [beadGrid[y][size-1-x], beadGrid[y][x]];
        }
      }
      drawBeadGrid();
    }

    function onMouseDown(e) {
      const rect = previewCanvas.getBoundingClientRect();
      const x = Math.floor((e.clientX-rect.left) / (previewCanvas.width/currentGridSize));
      const y = Math.floor((e.clientY-rect.top) / (previewCanvas.height/currentGridSize));
      if (currentTool === 'select') { selectionStart = {x,y}; selectionEnd = {x,y}; }
      if (currentTool === 'color') { saveHistory(); beadGrid[y][x] = getColorIndex(selectedColorInput.value.toUpperCase()); drawBeadGrid(); }
      if (currentTool === 'erase') { saveHistory(); beadGrid[y][x] = null; drawBeadGrid(); }
    }

    function onMouseMove(e) {
      const rect = previewCanvas.getBoundingClientRect();
      const x = Math.floor((e.clientX-rect.left) / (previewCanvas.width/currentGridSize));
      const y = Math.floor((e.clientY-rect.top) / (previewCanvas.height/currentGridSize));
      if (currentTool === 'select' && selectionStart) { selectionEnd = {x,y}; drawSelection(); }
      if (currentTool === 'color' && e.buttons === 1) { beadGrid[y][x] = getColorIndex(selectedColorInput.value.toUpperCase()); drawBeadGrid(); }
      if (currentTool === 'erase' && e.buttons === 1) { beadGrid[y][x] = null; drawBeadGrid(); }
    }

    function onMouseUp() {
      if (currentTool === 'select' && selectionStart && selectionEnd) {
        saveHistory();
        const minX = Math.min(selectionStart.x, selectionEnd.x), maxX = Math.max(selectionStart.x, selectionEnd.x);
        const minY = Math.min(selectionStart.y, selectionEnd.y), maxY = Math.max(selectionStart.y, selectionEnd.y);
        for (let y = minY; y <= maxY; y++) for (let x = minX; x <= maxX; x++) beadGrid[y][x] = getColorIndex(selectedColorInput.value.toUpperCase());
        drawBeadGrid();
      }
      selectionStart = null; selectionEnd = null;
    }

    function drawSelection() {
      drawBeadGrid();
      if (!selectionStart || !selectionEnd) return;
      const cell = previewCanvas.width / currentGridSize;
      const minX = Math.min(selectionStart.x, selectionEnd.x), maxX = Math.max(selectionStart.x, selectionEnd.x);
      const minY = Math.min(selectionStart.y, selectionEnd.y), maxY = Math.max(selectionStart.y, selectionEnd.y);
      ctx.strokeStyle = '#00FF00'; ctx.lineWidth = 2;
      ctx.strokeRect(minX*cell, minY*cell, (maxX-minX+1)*cell, (maxY-minY+1)*cell);
    }

    function addDraggableText() {
      const text = textInput.value.trim();
      if (!text) return;
      const div = document.createElement('div');
      div.className = 'absolute cursor-move bg-white bg-opacity-80 px-2 py-1 rounded shadow text-sm pointer-events-auto';
      div.textContent = text;
      div.style.left = '50%'; div.style.top = '10px'; div.style.transform = 'translateX(-50%)';
      div.addEventListener('mousedown', e => {
        draggedText = div;
        dragOffset.x = e.clientX - div.getBoundingClientRect().left;
        dragOffset.y = e.clientY - div.getBoundingClientRect().top;
      });
      textOverlay.appendChild(div);
    }

    window.addEventListener('mousemove', e => {
      if (!draggedText) return;
      const rect = $('previewWrapper').getBoundingClientRect();
      draggedText.style.left = (e.clientX-rect.left-dragOffset.x) + 'px';
      draggedText.style.top = (e.clientY-rect.top-dragOffset.y) + 'px';
      draggedText.style.transform = 'none';
    });
    window.addEventListener('mouseup', () => { draggedText = null; });

    function export4K() {
      const c = document.createElement('canvas');
      const scale = 4;
      c.width = previewCanvas.width * scale;
      c.height = previewCanvas.height * scale;
      const t = c.getContext('2d');
      t.scale(scale, scale);
      t.drawImage(previewCanvas, 0, 0);
      textOverlay.querySelectorAll('div').forEach(div => {
        const rect = div.getBoundingClientRect();
        const wRect = $('previewWrapper').getBoundingClientRect();
        t.font = '16px Arial';
        t.fillStyle = '#000';
        t.fillText(div.textContent, (rect.left-wRect.left)/scale, (rect.top-wRect.top)/scale + 12);
      });
      const a = document.createElement('a');
      a.download = `bead_pattern_${Date.now()}.png`;
      a.href = c.toDataURL('image/png');
      a.click();
    }

    init();
  </script>
